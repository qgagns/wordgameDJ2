    const Sound = {
        ctx: null,
        muted: safeStorageGet('spirit_muted', false),
        init() {
            if (!this.ctx) {
                const AudioContext = window.AudioContext || window.webkitAudioContext;
                if (AudioContext) this.ctx = new AudioContext();
            }
            if (this.ctx && this.ctx.state === 'suspended') {
                this.ctx.resume();
            }
        },
        toggleMute() {
            this.muted = !this.muted;
            safeStorageSet('spirit_muted', this.muted);
            return this.muted;
        },
        playTone(freq, type, duration, startVol = 0.2, endVol = 0) {
            if (this.muted) return;
            this.init();
            if (!this.ctx) return;
            try {
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = type;
                osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
                gain.gain.setValueAtTime(startVol, this.ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(Math.max(0.0001, endVol), this.ctx.currentTime + duration);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + duration);
            } catch(e) {}
        },

                { f: 783.99, d: 0.14 },

    function parseTextToWorlds(rawText) {
        if (!rawText) return [];
        const lines = rawText.split('\n');
        const maps = [];
        let currentMap = null;

        lines.forEach(line => {
            line = line.trim();
            if (!line) return;

            if (line.startsWith('[') && line.includes(']')) {
                let rawHeader = line.substring(1, line.indexOf(']')).trim();
                let mapName = rawHeader;
                let mapDesc = "";
                if (rawHeader.includes('|')) {
                    const parts = rawHeader.split('|');
                    mapName = parts[0].trim();
                    mapDesc = parts.slice(1).join('|').trim();
                }
                const mapId = 'map_' + Math.abs(hashCode(mapName));
                currentMap = { id: mapId, name: mapName, desc: mapDesc, words: [] };
                maps.push(currentMap);
                return;
            }

            if (line.startsWith('#') || line.startsWith('//')) {
                if (currentMap && !currentMap.desc) {
                    currentMap.desc = line.replace(/^[#/]+\s*/, '').trim();
                }
                return;
            }

            if (currentMap) {
                let eng = "", kor = "";
                if (line.includes(':')) {
                    const p = line.split(':');
                    eng = p[0].trim();
                    kor = p.slice(1).join(':').trim();
                } else if (line.includes('\t')) {
                    const p = line.split('\t');
                    eng = p[0].trim();
                    kor = p.slice(1).join(' ').trim();
                } else if (line.includes(',')) {
                    const p = line.split(',');
                    eng = p[0].trim();
                    kor = p.slice(1).join(',').trim();
                }
                if (eng && kor) currentMap.words.push({ eng, kor });
            }
        });

        maps.forEach(m => {
            if (!m.desc) {
                const theme = getMapTheme(m.name, m.id);
                m.desc = theme.defaultDesc;
            }
        });

        return maps.filter(m => m.words.length >= 4);
    }

    function hashCode(str) {
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            hash = ((hash << 5) - hash) + str.charCodeAt(i);
            hash |= 0;
        }
        return hash;
    }

    function parseOldWords(txt) {
        return txt.split('\n').map(l => {
            const p = l.split(':');
            return p.length >= 2 ? { eng: p[0].trim(), kor: p.slice(1).join(':').trim() } : null;
        }).filter(Boolean);
    }

    function getMapSectors(map) {
        const sectors = [];
        const total = map.words.length;
        const count = Math.ceil(total / WORDS_PER_SECTOR) || 1;
        for (let i = 0; i < count; i++) {
            const start = i * WORDS_PER_SECTOR;
            const end = Math.min(start + WORDS_PER_SECTOR, total);
            const sectorWords = map.words.slice(start, end);
            sectors.push({
                id: `${map.id}_sec_${i}`,
                index: i,
                name: `제${i + 1}구역`,
                rangeText: `정령 ${start + 1}~${end}번 (${sectorWords.length}마리)`,
                words: sectorWords
            });
        }
        return sectors;
    }

    function safeStorageGet(key, defaultVal) {
        try {
            const item = localStorage.getItem(key);
            if (!item || item === "undefined" || item === "null") return defaultVal;
            return JSON.parse(item);
        } catch (e) { return defaultVal; }
    }

    function safeStorageSet(key, val) {
        try { localStorage.setItem(key, JSON.stringify(val)); } catch (e) {}
    }

    function getMapCalculatedTier(mapId, wordsLength, contractedCount) {
        let percent = wordsLength > 0 ? Math.round((contractedCount / wordsLength) * 100) : 0;
        const bossDefeated = isMapBossDefeated(mapId);

        if (percent >= 100 && !bossDefeated) {
            return {
                tier: { level: 3, name: "수호령의 시련 대기 ⚔️", range: "구역 정복 완료", bg: "#059669", color: "#ecfdf5" },
                percent: 99,
                isConquered: false
            };
        }

        if (percent === 0) return { tier: mapMasteryTiers[0], percent: 0, isConquered: false };
        if (percent < 30) return { tier: mapMasteryTiers[1], percent, isConquered: false };
        if (percent < 60) return { tier: mapMasteryTiers[2], percent, isConquered: false };
        if (percent < 100 || !bossDefeated) return { tier: mapMasteryTiers[3], percent, isConquered: false };
        return { tier: mapMasteryTiers[4], percent: 100, isConquered: true };
    }

    function getGlobalTotalStats() {
        let totalWords = 0;
        let totalContracted = 0;
        let allMapsConquered = true;

        worldMaps.forEach(map => {
            totalWords += map.words.length;
            const savedMastery = (map.id === currentMapId && typeof wordMastery !== 'undefined')
                ? wordMastery
                : safeStorageGet(`spirit_mastery_${map.id}`, {});

            let mapContracted = 0;
            map.words.forEach(w => {
                if ((savedMastery[w.eng] || 0) >= 3) mapContracted++;
            });
            totalContracted += mapContracted;

            const bossDefeated = isMapBossDefeated(map.id);
            if (mapContracted < map.words.length || !bossDefeated) {
                allMapsConquered = false;
            }
        });
        return { totalWords, totalContracted, allMapsConquered };
    }

/* ========================================================
       🧠 스마트 단어 출제 엔진 (중복 원천 차단 & 쿨다운 순환)
       ======================================================== */
    function setupRound() {
        roundQuestions = [];
        currentQuestionIdx = 0;
        roundCorrectCount = 0;

        const pickedWords = []; // 이번 5문제에 들어갈 '서로 다른' 단어들

        // 1. [1순위] 오답 복습 단어 (최대 2마리, 이번 라운드 중복 방지)
        const reviewCandidates = [...wrongPool].sort(() => Math.random() - 0.5);
        for (const rw of reviewCandidates) {
            if (pickedWords.length >= 2) break;
            const wordObj = rawWords.find(w => w.eng === rw.engWord);
            if (wordObj && !pickedWords.some(p => p.eng === wordObj.eng)) {
                pickedWords.push({ ...wordObj, isReview: true });
            }
        }

        // 2. [2순위] 아직 마스터 안 된 단어 추출 (wordMastery < 3)
        const unmastered = rawWords.filter(w => (wordMastery[w.eng] || 0) < 3 && !pickedWords.some(p => p.eng === w.eng));

        // 🌟 쿨다운 적용: 방금 직전 라운드에 풀었던 단어는 잠시 휴식(후순위)
        const freshUnmastered = unmastered.filter(w => !recentRoundWords.includes(w.eng));
        const cooldownUnmastered = unmastered.filter(w => recentRoundWords.includes(w.eng));

        // 0회 정답 ➔ 1회 정답 ➔ 2회 정답 순으로 정렬
        function sortByMastery(list) {
            const g0 = list.filter(w => (wordMastery[w.eng] || 0) === 0).sort(() => Math.random() - 0.5);
            const g1 = list.filter(w => (wordMastery[w.eng] || 0) === 1).sort(() => Math.random() - 0.5);
            const g2 = list.filter(w => (wordMastery[w.eng] || 0) === 2).sort(() => Math.random() - 0.5);
            return [...g0, ...g1, ...g2];
        }

        const candidateQueue = [
            ...sortByMastery(freshUnmastered),
            ...sortByMastery(cooldownUnmastered)
        ];

        for (const w of candidateQueue) {
            if (pickedWords.length >= QUESTIONS_PER_ROUND) break;
            pickedWords.push({ ...w, isReview: false });
        }

        // 3. [3순위] 종반부 스마트 보충 (미완료 단어가 1~2개뿐이라 5문제가 안 채워질 때)
        // 🌟 이미 마스터한 18개 단어 중 직전 라운드에 안 나왔던 단어들을 '중복 없이 1개씩만' 차출
        if (pickedWords.length < QUESTIONS_PER_ROUND) {
            const mastered = rawWords.filter(w => !pickedWords.some(p => p.eng === w.eng));
            const freshMastered = mastered.filter(w => !recentRoundWords.includes(w.eng)).sort(() => Math.random() - 0.5);
            const fallbackMastered = mastered.filter(w => recentRoundWords.includes(w.eng)).sort(() => Math.random() - 0.5);
            const fillerQueue = [...freshMastered, ...fallbackMastered];

            for (const fw of fillerQueue) {
                if (pickedWords.length >= QUESTIONS_PER_ROUND) break;
                pickedWords.push({ ...fw, isReview: false });
            }
        }

        // 4. 다음 라운드 쿨타임을 위해 이번에 출제된 단어 5개 기억
        recentRoundWords = pickedWords.map(w => w.eng);

        // 5. [1단어 1문제 원칙] 한 단어당 딱 1문제씩만 생성 (E2K or K2E 50% 랜덤 결정, 동일 라운드 중복 원천 차단!)
        for (const w of pickedWords) {
            const type = Math.random() < 0.5 ? "E2K" : "K2E";
            const qId = `${currentMapId}_${type}_${w.eng}_r${currentRound}`;
            roundQuestions.push({
                id: qId,
                wordKey: w.eng,
                engWord: w.eng,
                korMeaning: w.kor,
                type: type,
                prompt: type === "E2K" ? w.eng : w.kor,
                correct: type === "E2K" ? w.kor : w.eng,
                pool: type === "E2K" ? rawWords.map(item => item.kor) : rawWords.map(item => item.eng),
                isReview: !!w.isReview
            });
        }

        // 6. 문제 순서 무작위 셔플 후 첫 문제 출제
        roundQuestions.sort(() => Math.random() - 0.5);
        loadQuestion();
    }

let questionStartTime = 0; // 🌟 문제 출제 시점 기록

    function loadQuestion() {
        if (currentQuestionIdx >= QUESTIONS_PER_ROUND) {
            showRoundResult();
            return;
        }

        questionStartTime = Date.now(); // 🌟 문제 표시되는 순간 타이머 스타트!

        document.getElementById("groveInfoDisplay").innerText = `🌿 탐사 ${currentQuestionIdx + 1} / ${QUESTIONS_PER_ROUND}`;

        const q = roundQuestions[currentQuestionIdx];
        document.getElementById("wordDisplay").innerText = q.prompt;
        document.getElementById("reviewTag").style.display = q.isReview ? "inline-block" : "none";

        const count = wordMastery[q.wordKey] || 0;
        const mInfo = getMasteryInfo(count);
        const mTag = document.getElementById("masteryTag");
        mTag.innerText = mInfo.stage === 3 ? "영원의 계약 🌟" : (mInfo.stage === 2 ? "따스한 유대 🍃" : (mInfo.stage === 1 ? "서먹한 대면 🐾" : "야생의 정령 🌿"));
        mTag.style.backgroundColor = mInfo.bg;
        mTag.style.color = mInfo.color;
        mTag.style.border = `1px solid ${mInfo.stage === 3 ? '#eab308' : '#cbd5e1'}`;

        if (streakCount >= 2) {
            streakBadge.style.display = "block";
            if (streakCount >= 5) streakBadge.innerText = `⚡ ${streakCount}연속 FEVER!`;
            else streakBadge.innerText = `🔥 ${streakCount}연속 교감!`;
        } else {
            streakBadge.style.display = "none";
        }

        const wrongChoices = q.pool.filter(item => item !== q.correct).sort(() => Math.random() - 0.5).slice(0, 3);
        const choices = [...wrongChoices, q.correct].sort(() => Math.random() - 0.5);

            btn.className = "choice-btn";
            btn.innerText = choice;
            btn.onclick = () => selectAnswer(choice, q);
            choicesGrid.appendChild(btn);
        });
    }

function selectAnswer(selectedAnswer, q) {
        const isCorrect = (selectedAnswer === q.correct);
        const elapsedSec = parseFloat(((Date.now() - questionStartTime) / 1000).toFixed(1)); // 반응 시간 계산
        const isResonance = isCorrect && (elapsedSec <= 1.5); // 🌟 1.5초 이내 정답 여부

        const fbModal = document.getElementById("feedbackModal");
        const fbTitle = document.getElementById("fbTitle");
        const fbWord = document.getElementById("fbWord");
        const fbMeaning = document.getElementById("fbMeaning");
        const fbArtBox = document.getElementById("fbArtBox");
        const fbContractNotice = document.getElementById("fbContractNotice");
        const fbResonanceNotice = document.getElementById("fbResonanceNotice");

        fbArtBox.innerHTML = getFeedbackArtSVG(isCorrect);
        fbWord.innerText = q.engWord;
        fbMeaning.innerText = q.korMeaning;

        const beforeTierIdx = getGlobalTier().tierIndex;
        const currentCountBefore = wordMastery[q.wordKey] || 0;

        if (isCorrect) {
            streakCount++;
            Sound.correct(streakCount);
            triggerSpiritTameEffect();

            // 콤보 칭호 체크
            if (streakCount > gameRecords.maxStreak) {
                gameRecords.maxStreak = streakCount;
                saveRecords();
            }
            if (streakCount >= 7) unlockTitle("combo_7");
            if (streakCount >= 15) unlockTitle("combo_15");

            roundCorrectCount++;

            // 🌟 [찰나의 공명 밸런스]: 1.5초 컷이면 숙련도 +2, 일반 정답이면 +1
            const gain = isResonance ? 2 : 1;
            const newCount = Math.min(3, currentCountBefore + gain);
            wordMastery[q.wordKey] = newCount;

            // 🌟 찰나의 공명 연출 (반응 속도 초단위 표기!)
            if (isResonance) {
                fbResonanceNotice.style.display = "block";
                fbResonanceNotice.innerHTML = `⚡ <b>[찰나의 공명 (${elapsedSec}초)!]</b> 본능적 교감으로 숙련도 <b>+2</b> 급상승!`;
            } else {
                fbResonanceNotice.style.display = "none";
            }

            // 영원의 계약(3회 완벽 달성) 처리
            if (newCount === 3) {
                fbContractNotice.style.display = "block";
                fbTitle.innerText = isResonance ? "🌟 찰나의 영원 계약 성립! 🌟" : "🌟 영원의 계약 성립! 🌟";
                fbTitle.style.color = "#d97706";
                Sound.fanfare();

                const { totalWords, totalContracted } = getGlobalTotalStats();
                if (totalContracted >= 1) unlockTitle("first_bond");
                if (totalContracted >= 10) unlockTitle("ten_bonds");
                if (totalContracted >= totalWords) unlockTitle("all_master");
            } else {
                fbContractNotice.style.display = "none";
                if (isResonance) {
                    fbTitle.innerText = `⚡ 찰나의 공명 교감 성공! (+2) ✨`;
                    fbTitle.style.color = "#0284c7";
                } else {
                    fbTitle.innerText = streakCount >= 2 ? `교감 성공! (${streakCount}연속) ✨` : "교감 성공! ✨";
                    fbTitle.style.color = "#15803d";
                }
            }

            fbMeaning.className = "fb-meaning";

            if (q.isReview) {
                wrongPool = wrongPool.filter(item => item.id !== q.id);
            }
            saveCurrentMapState();

            const afterTierObj = getGlobalTier();
            if (afterTierObj.tierIndex > beforeTierIdx) {
                pendingTierUp = {
                    oldTier: afterTierObj.tiers[beforeTierIdx],
                    newTier: afterTierObj.tier
                };
            }
        } else {
            streakCount = 0;
            Sound.wrong();
            triggerPlayerDamageEffect();

            // 오답 칭호 체크
            gameRecords.totalWrong++;
            saveRecords();
            if (gameRecords.totalWrong >= 20) unlockTitle("wrong_20");
            if (gameRecords.totalWrong >= 50) unlockTitle("wrong_50");
            
            fbResonanceNotice.style.display = "none";
            fbContractNotice.style.display = "none";
            fbTitle.innerText = "교감 실패 (정령이 숨었습니다) 🍂";
            fbTitle.style.color = "#b91c1c";
            fbMeaning.className = "fb-meaning wrong-case";

            if (!wrongPool.some(item => item.id === q.id)) {
                wrongPool.push(q);
            }
            saveCurrentMapState();
        }

        updateInGameTiers();
        fbModal.style.display = "flex";
    }

    function runBattleCountdown(callback) {
        const overlay = document.getElementById("battleCountdownOverlay");
        const display = document.getElementById("countdownNumDisplay");
        overlay.style.display = "flex";

        let count = 3;
        display.innerText = count;
        display.style.animation = 'none';
        display.offsetHeight;
        display.style.animation = null;
        Sound.countdownTick();

        const countInterval = setInterval(() => {
            count--;
            if (count > 0) {
                display.innerText = count;
                display.style.animation = 'none';
                display.offsetHeight;
                display.style.animation = null;
                Sound.countdownTick();
            } else if (count === 0) {
                display.innerText = "START!";
                display.style.color = "#10b981";
                display.style.animation = 'none';
                display.offsetHeight;
                display.style.animation = null;
                Sound.countdownStart();
            } else {
                clearInterval(countInterval);
                overlay.style.display = "none";
                display.style.color = "#fbbf24";
                callback();
            }
        }, 900);
    }

    function loadNextBossQuestion() {
        if (bossHp <= 0) {
            clearInterval(bossTimerInterval);
            document.getElementById("bossScreen").style.display = "none";

            if (bossMode === "mid") {
                if (playerShields === 3) unlockTitle("mid_no_dmg");
                if (hadBossDefeatBefore) unlockTitle("phoenix");
                hadBossDefeatBefore = false;
                handleMidBossVictory();
            } else {
                safeStorageSet('spirit_exam_cleared', true);
                Sound.fanfare();

                // 🌟 마왕 토벌 기록 및 칭호 갱신
                const elapsedSec = parseFloat(((Date.now() - bossStartTime) / 1000).toFixed(1));
                if (!gameRecords.fastestBossTime || elapsedSec < gameRecords.fastestBossTime) {
                    gameRecords.fastestBossTime = elapsedSec;
                }
                gameRecords.bossClearCount++;

                if (elapsedSec <= 25.0) unlockTitle("speed_demon");
                if (playerShields === 3) {
                    gameRecords.bossNoDamage = true;
                    unlockTitle("legend_ok");
                }
                if (playerShields === 1) unlockTitle("clutch_win");
                if (gameRecords.bossClearCount >= 3) unlockTitle("boss_3");
                if (hadBossDefeatBefore) unlockTitle("phoenix");
                hadBossDefeatBefore = false;

                saveRecords();
                document.getElementById("bossVictoryModal").style.display = "flex";
            }
            return;
        }

        if (playerShields <= 0) {
            clearInterval(bossTimerInterval);
            Sound.damage();
            hadBossDefeatBefore = true;
            gameRecords.bossDefeatCount++;
            unlockTitle("mental_break");
            saveRecords();

            if (bossMode === "mid") {
                alert("🛡️ 수호령의 시험에 패배했습니다...\n영지의 모든 정령들을 복습한 후 다시 도전하십시오!");
                document.getElementById("bossScreen").style.display = "none";
                openMapLobby(currentMapId);
            } else {
                alert("💥 마왕 『EXAM』의 시련에 결계가 무너졌습니다...\n재정비 후 다시 도전하십시오!");
                document.getElementById("bossScreen").style.display = "none";
                renderTitleScreen();
            }
            return;
        }

        isBossChecking = false;
        const targetWord = bossWordPool[Math.floor(Math.random() * bossWordPool.length)];
        const isE2K = Math.random() < 0.5;

        currentBossQuestion = {
            prompt: isE2K ? targetWord.eng : targetWord.kor,
            correct: isE2K ? targetWord.kor : targetWord.eng,
            pool: isE2K ? bossWordPool.map(w => w.kor) : bossWordPool.map(w => w.eng)
        };

        document.getElementById("bossWordDisplay").innerText = currentBossQuestion.prompt;

        const wrongChoices = currentBossQuestion.pool.filter(item => item !== currentBossQuestion.correct).sort(() => Math.random() - 0.5).slice(0, 3);
        const choices = [...wrongChoices, currentBossQuestion.correct].sort(() => Math.random() - 0.5);

        const choicesGrid = document.getElementById("bossChoicesGrid");
        choicesGrid.innerHTML = "";
        choices.forEach(choice => {
            const btn = document.createElement("button");
            btn.className = "choice-btn";
            btn.innerText = choice;
            btn.onclick = () => selectBossAnswer(btn, choice);
            choicesGrid.appendChild(btn);
        });

        startBossTimer();
    }


    function startBossTimer() {
        clearInterval(bossTimerInterval);
        bossTimeLeft = bossTimeLimit;
        const timerBar = document.getElementById("bossTimerBar");
        timerBar.className = "boss-timer-bar";
        timerBar.style.width = "100%";

        bossTimerInterval = setInterval(() => {
            bossTimeLeft -= 0.1;
            const percent = (bossTimeLeft / bossTimeLimit) * 100;
            timerBar.style.width = `${Math.max(0, percent)}%`;

            if (bossTimeLeft <= 2.0) {
                timerBar.classList.add("urgent");
            }

            if (bossTimeLeft <= 0) {
                clearInterval(bossTimerInterval);
                handleBossTimeOut();
            }
        }, 100);
    }

    function selectBossAnswer(btn, choice) {
        if (isBossChecking) return;
        isBossChecking = true;
        clearInterval(bossTimerInterval);

        if (choice === currentBossQuestion.correct) {
            Sound.hit();
            btn.classList.add("correct");
            bossHp--;
            // 🌟 정답 시 보스에게 참격 + 흔들림 + 데미지 텍스트 (보스 HP가 2 이하일 땐 크리티컬 더블 참격!)
            triggerBossHitEffect(bossHp <= 2);
        } else {
            Sound.damage();
            btn.classList.add("wrong");
            playerShields--;
            // 🌟 오답 시 플레이어 화면 지진 + 붉은 번쩍임
            triggerPlayerDamageEffect();
            document.querySelectorAll("#bossChoicesGrid .choice-btn").forEach(b => {
                if (b.innerText === currentBossQuestion.correct) b.classList.add("correct");
            });
        }

        updateBossUI();
        setTimeout(loadNextBossQuestion, 800);
    }

    function handleBossTimeOut() {
        if (isBossChecking) return;
        isBossChecking = true;
        Sound.damage();
        playerShields--;
        triggerPlayerDamageEffect();

        // 타임오버 칭호 체크
        gameRecords.timeOutCount++;
        saveRecords();
        if (gameRecords.timeOutCount >= 3) unlockTitle("time_out");

        updateBossUI();

        document.querySelectorAll("#bossChoicesGrid .choice-btn").forEach(b => {
            if (b.innerText === currentBossQuestion.correct) b.classList.add("correct");
        });

        setTimeout(loadNextBossQuestion, 1000);
    }

    function openWordBook(fromTitle = false) {
        Sound.click();
        const inGameView = document.getElementById("inGameBookView");
        const titleView = document.getElementById("titleBookView");

        if (fromTitle) {
            inGameView.style.display = "none";
            titleView.style.display = "block";

            const { totalWords, totalContracted } = getGlobalTotalStats();
            const totalPercent = totalWords > 0 ? Math.round((totalContracted / totalWords) * 100) : 0;

            document.getElementById("titleOverallPercentText").innerText = `${totalPercent}%`;
            document.getElementById("titleOverallGaugeBar").style.width = `${totalPercent}%`;
            document.getElementById("titleTotalWordsSummary").innerText = `계약: ${totalContracted} / ${totalWords}마리`;

            const overviewList = document.getElementById("titleWorldMapOverviewList");
            overviewList.innerHTML = "";

            worldMaps.forEach(map => {
                const savedMastery = safeStorageGet(`spirit_mastery_${map.id}`, {});
                let mapContracted = 0;
                map.words.forEach(w => {
                    if ((savedMastery[w.eng] || 0) >= 3) mapContracted++;
                });
                const { tier: mTier, percent } = getMapCalculatedTier(map.id, map.words.length, mapContracted);

                const card = document.createElement("div");
                card.className = "world-summary-card";
                card.innerHTML = `
                    <div class="world-summary-top">
                        <span>${map.name}</span>
                        <span class="tag" style="background:${mTier.bg}; color:${mTier.color}; padding:2px 7px; font-size:10px;">${mTier.name}</span>
                    </div>
                    <div style="font-size:10px; color:#8c6d46; margin-bottom:2px;">${map.desc}</div>
                    <div class="overall-gauge-container" style="height:6px; margin:4px 0 2px 0;">
                        <div class="overall-gauge-bar" style="width:${percent}%;"></div>
                    </div>
                    <div style="font-size:10px; color:#785a3e; text-align:right; font-weight:700;">
                        ${mapContracted} / ${map.words.length}마리 (${percent}%)
                    </div>
                `;
                overviewList.appendChild(card);
            });
        } else {
            inGameView.style.display = "block";
            titleView.style.display = "none";

            const listContainer = document.getElementById("wordListContainer");
            listContainer.innerHTML = "";

            const activeMap = worldMaps.find(m => m.id === currentMapId) || worldMaps[0];
            const currentWords = activeMap.words;
            const currentSavedMastery = safeStorageGet(`spirit_mastery_${activeMap.id}`, {});

            let boundCount = 0;
            currentWords.forEach(item => {
                const count = currentSavedMastery[item.eng] || 0;
                if (count >= 3) boundCount++;
                const mInfo = getMasteryInfo(count);

                const row = document.createElement("div");
                row.className = `word-item-clean stage-${mInfo.stage}`;
                row.innerHTML = `
                    <div class="word-text-group">
                        <span class="word-main-eng">${item.eng}</span>
                        <span class="word-sub-kor">${item.kor}</span>
                    </div>
                    <span class="word-mini-status" style="background:${mInfo.bg}; color:${mInfo.color}; border: 1px solid ${mInfo.stage === 3 ? '#eab308' : '#cbd5e1'};">
                        ${mInfo.text}
                    </span>
                `;
                listContainer.appendChild(row);
            });

            const overallPercent = currentWords.length > 0 ? Math.round((boundCount / currentWords.length) * 100) : 0;
            document.getElementById("overallPercentageText").innerText = `${boundCount} / ${currentWords.length}마리 (${overallPercent}%)`;
            document.getElementById("overallGaugeBar").style.width = overallPercent + "%";
            document.getElementById("mapBookTitleText").innerText = `🌟 ${activeMap.name} 교감도`;
        }

        const tierListContainer = document.getElementById("tierRoadmapList");
        tierListContainer.innerHTML = "";
        const { tier: currentTier, totalWords, tiers } = getGlobalTier();

        tiers.forEach((t, idx) => {
            const isCurrent = (t.name === currentTier.name);
            const item = document.createElement("div");
            item.className = `timeline-item ${isCurrent ? 'current-tier' : ''}`;
            let reqText = (idx === tiers.length - 1) ? `전체 ${totalWords}마리 완벽 계약` : `${t.req}마리 계약 달성`;
            if (idx === 0) reqText = "시작 (0마리)";

            item.innerHTML = `
                <span style="font-weight:bold; color:${isCurrent ? '#b45309' : '#334155'};">${t.name}</span>
                <span style="font-size:11px; color:${isCurrent ? '#b45309' : '#64748b'}; font-weight:700;">
                    ${isCurrent ? '📍 [현재 계급]' : reqText}
                </span>
            `;
            tierListContainer.appendChild(item);
        });

        const mapRoadmapContainer = document.getElementById("mapMasteryRoadmapList");
        mapRoadmapContainer.innerHTML = "";
        
        const activeMap = worldMaps.find(m => m.id === currentMapId) || worldMaps[0];
        const savedMastery = safeStorageGet(`spirit_mastery_${activeMap.id}`, {});
        let mapContracted = 0;
        activeMap.words.forEach(w => {
            if ((savedMastery[w.eng] || 0) >= 3) mapContracted++;
        });
        const { tier: currentMapTier } = getMapCalculatedTier(currentMapId, activeMap.words.length, mapContracted);

        mapMasteryTiers.forEach((mt) => {
            const isCurrent = (mt.name === currentMapTier.name);
            const item = document.createElement("div");
            item.className = `timeline-item ${isCurrent ? 'current-tier' : ''}`;
            item.innerHTML = `
                <span style="font-weight:bold; color:${isCurrent ? '#b45309' : '#334155'};">${mt.name}</span>
                <span style="font-size:11px; color:${isCurrent ? '#b45309' : '#64748b'}; font-weight:700;">
                    ${isCurrent ? '📍 [현재 상태]' : mt.range}
                </span>
            `;
            mapRoadmapContainer.appendChild(item);
        });

        switchBookTab(1);
        document.getElementById("bookModal").style.display = "flex";
    }
